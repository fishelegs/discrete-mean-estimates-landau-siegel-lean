lemma lemma32_actual_interval_prefix_sum {D : ℕ} (χ : RealPrimitiveCharacter D) (N : ℕ) :
    lemma32BurgessIntervalSum χ 1 N=∑ n ∈ Finset.Icc 1 N, χ.evalNat n := by
  unfold lemma32BurgessIntervalSum
  have he (i : Fin N) : χ.chi (((1+(i.val : ℤ)) : ℤ) : ZMod D)=χ.evalNat (i.val+1) := by
    simp only [RealPrimitiveCharacter.evalNat, Int.cast_add, Int.cast_one,
      Int.cast_natCast, Nat.cast_add, Nat.cast_one, add_comm]
  simp_rw [he]
  rw [Fin.sum_univ_eq_sum_range (fun n => χ.evalNat (n+1)) N,
    Finset.range_eq_Ico, Finset.sum_Ico_add' (fun n => χ.evalNat n) 0 N 1]
  have hi : Finset.Ico (0+1) (N+1)=Finset.Icc 1 N := by
    ext n
    simp only [Finset.mem_Ico, Finset.mem_Icc]
    omega
  rw [hi]

lemma lemma32_actual_uniform_burgess_partial_sums (D : ℕ) (hD : 2≤D)
    (χ : RealPrimitiveCharacter D) (N : ℕ) :
    ‖∑ n ∈ Finset.Icc 1 N, χ.evalNat n‖ ≤
      (lemma32BurgessGlobalConstant*Real.exp ((25/128)*lemma23PaperL D))*(N : ℝ)^(1/2 : ℝ) := by
  letI : NeZero D := ⟨χ.modulus_ne_zero⟩

  rw [lemma32_actual_interval_prefix_sum] at hb
  simpa only [lemma32BurgessPower, lemma23PaperL, Real.sqrt_eq_rpow] using hb
