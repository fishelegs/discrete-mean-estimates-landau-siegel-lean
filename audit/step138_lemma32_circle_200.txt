lemma lemma32_actual_first_derivative_bound
    {D : ℕ} (χ : RealPrimitiveCharacter D) (hD : 1 < D)
    (hL : 2 ≤ Real.log (D : ℝ)) {s : ℂ}
    (hs : ‖s - 1‖ ≤ 1 / (4 * Real.log (D : ℝ))) :
    ‖deriv (dirichletLFunction χ) s‖ ≤
      16 * Real.exp 1 * Real.log (D : ℝ) ^ 2 := by
  let L := Real.log (D : ℝ)
  let R := 1 / (4 * L)
  have hLp : 0 < L := by dsimp [L]; linarith
  have hRp : 0 < R := by dsimp [R]; positivity
  have hR2 : R + R ≤ 1 / L := by
    dsimp [R]
    field_simp
    linarith
  have hLinv : 1 / L ≤ (1 : ℝ) / 2 :=
    one_div_le_one_div_of_le (by norm_num) hL
  have hdiff := differentiable_dirichletLFunction_of_one_lt_modulus χ hD
  have hboundary (z : ℂ) (hz : z ∈ sphere s R) :
      ‖dirichletLFunction χ z‖ ≤ 4 * Real.exp 1 * L := by
    have hdist : ‖z - s‖ = R := mem_sphere_iff_norm.mp hz
    have hz1 : ‖z - 1‖ ≤ 1 / L := by
      calc
        _ = ‖(z - s) + (s - 1)‖ := by congr 1; ring
        _ ≤ ‖z - s‖ + ‖s - 1‖ := norm_add_le _ _
        _ ≤ R + R := by
          rw [hdist]
          simpa [R, L, add_comm] using add_le_add_left hs R
        _ ≤ 1 / L := hR2
    have hre : |z.re - 1| ≤ 1 / L := by
      simpa using (Complex.abs_re_le_norm (z - 1)).trans hz1
    have hσ : 1 - 1 / L ≤ z.re := by
      have hr := (abs_le.mp hre).1
      linarith
    have hnorm : ‖z‖ ≤ 2 := by
      have hnorm' : ‖z‖ ≤ ‖z - 1‖ + 1 := by
        simpa using norm_add_le (z - 1) (1 : ℂ)
      linarith
    exact lemma55_actual_L_near_one_bound χ hD hL hσ hnorm
  calc
    _ ≤ (4 * Real.exp 1 * L) / R :=
      Complex.norm_deriv_le_of_forall_mem_sphere_norm_le hRp hdiff.diffContOnCl hboundary
    _ = 16 * Real.exp 1 * L^2 := by
      dsimp [R]
      field_simp
      ring

lemma lemma32_actual_L_difference_near_one {D : ℕ} (χ : RealPrimitiveCharacter D)
    (hD : 1 < D) (hL : 2 ≤ lemma23PaperL D) {s : ℂ}
    (hs : ‖s-1‖ ≤ 1/(4*lemma23PaperL D)) :
    ‖dirichletLFunction χ s-LAtOne χ‖ ≤ 16*Real.exp 1*lemma23PaperL D^2*‖s-1‖ := by
  let R := 1/(4*lemma23PaperL D)
  have hl0 : 0 < lemma23PaperL D := by linarith
  have hr : 0 < R := by dsimp [R]; exact one_div_pos.mpr (mul_pos (by norm_num) hl0)
  have hd := differentiable_dirichletLFunction_of_one_lt_modulus χ hD
  have hb (z : ℂ) (hz : z ∈ Metric.closedBall (1 : ℂ) R) :
      ‖deriv (dirichletLFunction χ) z‖ ≤ 16*Real.exp 1*lemma23PaperL D^2 :=
    lemma32_actual_first_derivative_bound χ hD hL (mem_closedBall_iff_norm.mp hz)
  have h1 : (1 : ℂ) ∈ Metric.closedBall (1 : ℂ) R := Metric.mem_closedBall_self hr.le
  have hsm : s ∈ Metric.closedBall (1 : ℂ) R := mem_closedBall_iff_norm.mpr hs
  simpa only [LAtOne] using Convex.norm_image_sub_le_of_norm_deriv_le
    (fun z _ => hd z) hb (convex_closedBall (1 : ℂ) R) h1 hsm

lemma lemma32_actual_value_at_one_small {D : ℕ} (χ : RealPrimitiveCharacter D)
    (hD : 1 < D) (hA : NormalizedAssumptionA χ) :
    ‖LAtOne χ‖ ≤ lemma23PaperL D^(-2022 : ℤ) := by
  have hA' : realLAtOne χ < lemma23PaperL D^(-2022 : ℤ) := by
    simpa only [NormalizedAssumptionA,AssumptionAWithConstant,lemma23PaperL,one_mul] using hA
  have hp : 0 < realLAtOne χ := realLAtOne_pos χ hD
  have hnorm : ‖LAtOne χ‖ = realLAtOne χ := by
    unfold LAtOne
    rw [show (1 : ℂ) = ((1 : ℝ) : ℂ) by simp,
      dirichletLFunction_real_eq_realLValue χ hD 1,Complex.norm_real,Real.norm_eq_abs]
    exact abs_of_pos hp
  exact hnorm.le.trans hA'.le

lemma lemma32_circle_small_value_scale_identity (L : ℝ) (hL : 0 < L) :
    L^2*L^(-2024 : ℤ) = L^(-2022 : ℤ) := by
  simpa only [Int.reduceAdd,zpow_ofNat] using (zpow_add₀ (ne_of_gt hL) 2 (-2024 : ℤ)).symm

lemma lemma32_actual_L_residue_disk_small {D : ℕ} (χ : RealPrimitiveCharacter D)
    (hD : 1 < D) (hL : 3 ≤ lemma23PaperL D) (hA : NormalizedAssumptionA χ)
    (s : ℂ) (hs : ‖s-1‖ ≤ lemma23PaperL D^(-2024 : ℤ)) :
    ‖dirichletLFunction χ s‖ ≤ (1+16*Real.exp 1)*lemma23PaperL D^(-2022 : ℤ) := by
  have hp := lemma32_residue_disk_parameters hL s hs
  have hd := lemma32_actual_L_difference_near_one χ hD (by linarith) hp.2.2
  have h1 := lemma32_actual_value_at_one_small χ hD hA
  have hl0 : 0 < lemma23PaperL D := by linarith
  calc
    _ ≤ ‖dirichletLFunction χ s-LAtOne χ‖+‖LAtOne χ‖ := by
      simpa only [sub_add_cancel] using norm_add_le (dirichletLFunction χ s-LAtOne χ) (LAtOne χ)
    _ ≤ 16*Real.exp 1*lemma23PaperL D^2*‖s-1‖+lemma23PaperL D^(-2022 : ℤ) := add_le_add hd h1
    _ ≤ 16*Real.exp 1*lemma23PaperL D^2*lemma23PaperL D^(-2024 : ℤ)+
        lemma23PaperL D^(-2022 : ℤ) := add_le_add
      (mul_le_mul_of_nonneg_left hs (by positivity)) (le_refl _)
    _ = (16*Real.exp 1)*(lemma23PaperL D^2*lemma23PaperL D^(-2024 : ℤ))+
        lemma23PaperL D^(-2022 : ℤ) := by ring
    _ = _ := by rw [lemma32_circle_small_value_scale_identity _ hl0]; ring
