lemma lemma32_weighted_fourth_moment_inequality {A : Type*} (s : Finset A)
    (w z : A → ℝ) (hw : ∀ a ∈ s, 0 ≤ w a) :
    (∑ a ∈ s, w a*z a)^4 ≤
      (∑ a ∈ s, w a)^2 * (∑ a ∈ s, (w a)^2) * (∑ a ∈ s, (z a)^4) := by
  have h1 : (∑ a ∈ s, w a*z a)^2 ≤
      (∑ a ∈ s, w a) * (∑ a ∈ s, w a*(z a)^2) := by
    apply Finset.sum_sq_le_sum_mul_sum_of_sq_le_mul s hw
      (fun a ha => mul_nonneg (hw a ha) (sq_nonneg _))
    intro a ha
    exact (by ring : (w a*z a)^2 = w a*(w a*(z a)^2)).le
  have h2 : (∑ a ∈ s, w a*(z a)^2)^2 ≤
      (∑ a ∈ s, (w a)^2)*(∑ a ∈ s, (z a)^4) := by
    apply Finset.sum_sq_le_sum_mul_sum_of_sq_le_mul s
      (fun a ha => sq_nonneg (w a)) (fun a ha => by positivity)
    intro a ha
    exact (by ring : (w a*(z a)^2)^2 = (w a)^2*(z a)^4).le
  calc
    _ = ((∑ a ∈ s, w a*z a)^2)^2 := by ring
    _ ≤ ((∑ a ∈ s, w a)*(∑ a ∈ s, w a*(z a)^2))^2 :=
      pow_le_pow_left₀ (sq_nonneg _) h1 2
    _ = (∑ a ∈ s, w a)^2*(∑ a ∈ s, w a*(z a)^2)^2 := mul_pow _ _ _
    _ ≤ (∑ a ∈ s, w a)^2*((∑ a ∈ s, (w a)^2)*(∑ a ∈ s, (z a)^4)) :=
      mul_le_mul_of_nonneg_left h2 (sq_nonneg _)
    _ = _ := by ring

lemma lemma32_actual_weighted_burgess_moment {D : ℕ} [NeZero D]
    (χ : RealPrimitiveCharacter D) (H : ℕ) (t : ZMod D → ℕ) :
    (∑ x : ZMod D, (t x : ℝ)*‖lemma32ResidueShortSum χ H x‖)^4 ≤
      (∑ x : ZMod D, (t x : ℝ))^2 * (∑ x : ZMod D, (t x : ℝ)^2) *
        lemma32BurgessFourthMoment χ H := by
  exact lemma32_weighted_fourth_moment_inequality Finset.univ
    (fun x => (t x : ℝ)) (fun x => ‖lemma32ResidueShortSum χ H x‖)
    (fun x hx => Nat.cast_nonneg _)
