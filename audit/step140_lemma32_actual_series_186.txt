  simpa only [add_zero] using (HasSum.even_add_odd
    (f := fun e : ℕ => (lemma32ActualCoefficient χ (p^e) : ℂ)*z^e) he ho)

noncomputable def lemma32LocalCorrection {D : ℕ} (χ : RealPrimitiveCharacter D)
    (p : ℕ) (z : ℂ) : ℂ :=
  (1-z)^8*(1-χ.evalNat p*z)^8*∑' e : ℕ, (lemma32ActualCoefficient χ (p^e) : ℂ)*z^e

lemma lemma32_local_correction_of_one {D p : ℕ} (χ : RealPrimitiveCharacter D)
    (hp : p.Prime) (h : χ.evalNat p = 1) (z : ℂ) (hz : ‖z‖ < 1) :
    lemma32LocalCorrection χ p z = (1-z)^11*(1+11*z+11*z^2+z^3) := by
  unfold lemma32LocalCorrection
  rw [h,(lemma32_actual_local_series_of_one χ hp h z hz).tsum_eq]
  have hn : 1-z ≠ 0 := sub_ne_zero.mpr (by intro h; rw [← h] at hz; norm_num at hz)
  simp only [one_mul]
  field_simp
  all_goals ring

lemma lemma32_local_correction_of_neg_one {D p : ℕ} (χ : RealPrimitiveCharacter D)
