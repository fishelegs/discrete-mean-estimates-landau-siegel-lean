lemma lemma32_burgess_full_collision_sum (D : ℕ) (M : ℤ) (A N : ℕ) :
    (∑ p : Fin A × Fin N, ∑ q : Fin A × Fin N,
      if (D : ℤ) ∣ lemma32BurgessDeterminant M (p.1.val+1) (q.1.val+1) p.2 q.2
      then (1 : ℝ) else 0) =
    ∑ a : Fin A, ∑ b : Fin A,
      (((Finset.univ : Finset (Fin N × Fin N)).filter
        (fun p => (D : ℤ) ∣ lemma32BurgessDeterminant M (a.val+1) (b.val+1) p.1 p.2)).card : ℝ) := by
  simp_rw [Fintype.sum_prod_type]
  apply Finset.sum_congr rfl
  intro a ha
  rw [Finset.sum_comm]
  apply Finset.sum_congr rfl
  intro b hb
  rw [← Fintype.sum_prod_type' (fun n m : Fin N =>
    if (D : ℤ) ∣ lemma32BurgessDeterminant M (a.val+1) (b.val+1) n m
    then (1 : ℝ) else 0)]
  rw [← Finset.sum_filter]
  simp

lemma lemma32_actual_burgess_multiplicity_energy_bound {D A N : ℕ} [NeZero D]
    (hsize : 2*A*N < D) (M : ℤ) :
    (∑ x : ZMod D, (lemma32BurgessMultiplicity D M A N x : ℝ)^2) ≤
      2*(A : ℝ)*(N : ℝ)*(1+Real.log (A : ℝ))+(A : ℝ)^2 := by
  rw [lemma32_actual_burgess_multiplicity_square_determinant]
  change (∑ p ∈ lemma32BurgessMultiplierPairs D A N,
    ∑ q ∈ lemma32BurgessMultiplierPairs D A N,
      if (D : ℤ) ∣ lemma32BurgessDeterminant M (p.1.val+1) (q.1.val+1) p.2 q.2
      then (1 : ℝ) else 0) ≤ _
  have hext : (∑ p ∈ lemma32BurgessMultiplierPairs D A N,
      ∑ q ∈ lemma32BurgessMultiplierPairs D A N,
        if (D : ℤ) ∣ lemma32BurgessDeterminant M (p.1.val+1) (q.1.val+1) p.2 q.2
        then (1 : ℝ) else 0) ≤
      ∑ p : Fin A × Fin N, ∑ q : Fin A × Fin N,
        if (D : ℤ) ∣ lemma32BurgessDeterminant M (p.1.val+1) (q.1.val+1) p.2 q.2
        then (1 : ℝ) else 0 := by
    calc
      _ ≤ ∑ p ∈ lemma32BurgessMultiplierPairs D A N, ∑ q : Fin A × Fin N,
          if (D : ℤ) ∣ lemma32BurgessDeterminant M (p.1.val+1) (q.1.val+1) p.2 q.2
          then (1 : ℝ) else 0 := by
        apply Finset.sum_le_sum
        intro p hp
        exact Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ _)
          (fun q hq hqp => by split_ifs <;> norm_num)
      _ ≤ _ := Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ _)
        (fun p hp hpp => Finset.sum_nonneg (fun q hq => by split_ifs <;> norm_num))
  rw [lemma32_burgess_full_collision_sum] at hext
  have hc : (∑ a : Fin A, ∑ b : Fin A,
      (((Finset.univ : Finset (Fin N × Fin N)).filter
        (fun p => (D : ℤ) ∣ lemma32BurgessDeterminant M (a.val+1) (b.val+1) p.1 p.2)).card : ℝ)) ≤
      ∑ a : Fin A, ∑ b : Fin A,
        ((N : ℝ)*((a.val+1).gcd (b.val+1) : ℝ)/
          ((max (a.val+1) (b.val+1) : ℕ) : ℝ)+1) := by
    apply Finset.sum_le_sum
    intro a ha
    apply Finset.sum_le_sum
    intro b hb
    exact lemma32_burgess_fixed_multiplier_max_count hsize
      (by omega) (by omega) (by omega) (by omega) M
  have he : (∑ a : Fin A, ∑ b : Fin A,
        ((N : ℝ)*((a.val+1).gcd (b.val+1) : ℝ)/
          ((max (a.val+1) (b.val+1) : ℕ) : ℝ)+1)) =
      (N : ℝ)*(∑ a : Fin A, ∑ b : Fin A,
        (((a.val+1).gcd (b.val+1) : ℝ)/
          ((max (a.val+1) (b.val+1) : ℕ) : ℝ)))+(A : ℝ)^2 := by
    simp_rw [mul_div_assoc, Finset.sum_add_distrib, ← Finset.mul_sum]
    simp only [Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]
    ring
  rw [he] at hc
  have hg := mul_le_mul_of_nonneg_left (lemma32_fin_multiplier_normalized_gcd_sum A)
    (Nat.cast_nonneg N : (0 : ℝ) ≤ (N : ℝ))
  exact (hext.trans hc).trans (by nlinarith)

lemma lemma32_actual_burgess_multiplicity_energy_log_bound {D A N : ℕ} [NeZero D]
    (hsize : 2*A*N < D) (hAN : A ≤ N) (hA : 1 ≤ A) (M : ℤ) :
    (∑ x : ZMod D, (lemma32BurgessMultiplicity D M A N x : ℝ)^2) ≤
      3*(A : ℝ)*(N : ℝ)*(1+Real.log (A : ℝ)) := by
  have hlog := Real.log_nonneg (Nat.one_le_cast.mpr hA : (1 : ℝ) ≤ (A : ℝ))
  have ha : (0 : ℝ) ≤ (A : ℝ) := Nat.cast_nonneg _
  have hn : (0 : ℝ) ≤ (N : ℝ) := Nat.cast_nonneg _
  have han : (A : ℝ) ≤ (N : ℝ) := Nat.cast_le.mpr hAN
  have hmul : (A : ℝ)^2 ≤ (A : ℝ)*(N : ℝ) := by nlinarith
  have hp : (0 : ℝ) ≤ (A : ℝ)*(N : ℝ)*Real.log (A : ℝ) :=
    mul_nonneg (mul_nonneg ha hn) hlog
  exact (lemma32_actual_burgess_multiplicity_energy_bound hsize M).trans (by nlinarith)
