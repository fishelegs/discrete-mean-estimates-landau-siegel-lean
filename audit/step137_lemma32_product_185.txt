lemma lemma32_actual_coefficient_mul {D m n : ℕ} (χ : RealPrimitiveCharacter D)
    (h : m.Coprime n) :
    lemma32ActualCoefficient χ (m*n) = lemma32ActualCoefficient χ m*lemma32ActualCoefficient χ n := by
  unfold lemma32ActualCoefficient lemma34Tau
  rw [(lemma31_actual_nu_multiplicative χ).map_mul_of_coprime h,norm_mul,
    (lemma34_tau_multiplicative 2).map_mul_of_coprime h]
  push_cast
  ring

lemma lemma32_square_geometric_hasSum (z : ℂ) (hz : ‖z‖ < 1) :
    HasSum (fun e : ℕ => (e+1 : ℂ)^2*z^e) ((1+z)/(1-z)^3) := by
  have h1 := hasSum_choose_mul_geometric_of_norm_lt_one 1 hz
  have h2 := (hasSum_choose_mul_geometric_of_norm_lt_one 2 hz).mul_left (2 : ℂ)
  have hn : 1-z ≠ 0 := sub_ne_zero.mpr (by intro h; rw [← h] at hz; norm_num at hz)
  convert h2.sub h1 using 1
  · funext e
    have hc := lemma32_rising_choose_two e
    simp only [Nat.choose_one_right,Nat.cast_add,Nat.cast_one]
    linear_combination -hc*z^e
  · field_simp
    ring

lemma lemma32_even_square_geometric_hasSum (z : ℂ) (hz : ‖z‖ < 1) :
    HasSum (fun e : ℕ => (2*e+1 : ℂ)^2*z^e) ((1+6*z+z^2)/(1-z)^3) := by
  have h0 := hasSum_geometric_of_norm_lt_one hz
  have h1 := (hasSum_choose_mul_geometric_of_norm_lt_one 1 hz).mul_left (8 : ℂ)
  have h2 := (hasSum_choose_mul_geometric_of_norm_lt_one 2 hz).mul_left (8 : ℂ)
  have hn : 1-z ≠ 0 := sub_ne_zero.mpr (by intro h; rw [← h] at hz; norm_num at hz)
  convert (h2.sub h1).add h0 using 1
  · funext e
    have hc := lemma32_rising_choose_two e
    simp only [Nat.choose_one_right,Nat.cast_add,Nat.cast_one]
    linear_combination -4*hc*z^e
  · field_simp
    ring
