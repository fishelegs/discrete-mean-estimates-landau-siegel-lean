example {D p : ℕ} (χ : RealPrimitiveCharacter D) (hp : p.Prime) (hpd : ¬p ∣ D)
    (z : ℂ) (hz : ‖z‖ < 1) :
    ‖(1-z)^8*(1-χ.evalNat p*z)^8*
      (∑' e : ℕ, ((‖lemma23NuArithmeticFunction χ (p^e)‖^2*
        (lemma34Tau 2 (p^e) : ℝ)^2 : ℝ) : ℂ)*z^e)-1‖ ≤ 7659*‖z‖^2 :=
    lemma32_local_correction_norm_of_not_dvd χ hp hpd z hz

example {D p : ℕ} (χ : RealPrimitiveCharacter D) (hp : p.Prime) (hpd : p ∣ D)
    (z : ℂ) (hz : ‖z‖ < 1) :
    (1-z)^8*(1-χ.evalNat p*z)^8*
      (∑' e : ℕ, ((‖lemma23NuArithmeticFunction χ (p^e)‖^2*
        (lemma34Tau 2 (p^e) : ℝ)^2 : ℝ) : ℂ)*z^e) = (1-z)^4*(1-z^2) :=
    lemma32_zero_correction_factorization χ hp
      (χ.evalNat_eq_zero_of_dvd_modulus hpd hp.ne_one) z hz

example {D : ℕ} (χ : RealPrimitiveCharacter D) (s : ℂ) (hs : 1/2 < s.re) :
    Multipliable (fun p : Nat.Primes => lemma32RegularLocalPolynomial χ p.val
      (Complex.exp (-s*(Real.log (p.val : ℝ) : ℂ)))) :=
    lemma32_regular_euler_product_multipliable χ s hs

example {D : ℕ} (χ : RealPrimitiveCharacter D) (σ : ℝ) (hσ : 1/2 < σ)
    (s : ℂ) (hs : σ ≤ s.re) :
    ‖∏' p : Nat.Primes, lemma32RegularLocalPolynomial χ p.val
      (Complex.exp (-s*(Real.log (p.val : ℝ) : ℂ)))‖ ≤
        Real.exp (7659*∑' p : Nat.Primes, (p.val : ℝ)^(-2*σ)) :=
    lemma32_regular_euler_product_bound χ σ hσ s hs

example {D : ℕ} (χ : RealPrimitiveCharacter D) :
    TendstoUniformlyOn (fun S : Finset Nat.Primes => fun s : ℂ =>
      ∏ p ∈ S, lemma32RegularLocalPolynomial χ p.val
        (Complex.exp (-s*(Real.log (p.val : ℝ) : ℂ))))
      (fun s : ℂ => ∏' p : Nat.Primes, lemma32RegularLocalPolynomial χ p.val
        (Complex.exp (-s*(Real.log (p.val : ℝ) : ℂ))))
      atTop {s : ℂ | 3/4 ≤ s.re} :=
    lemma32_regular_products_uniform_limit χ (3/4) (by norm_num)

example {D : ℕ} (χ : RealPrimitiveCharacter D) :
    AnalyticOnNhd ℂ (fun s : ℂ => ∏' p : Nat.Primes, lemma32RegularLocalPolynomial χ p.val
      (Complex.exp (-s*(Real.log (p.val : ℝ) : ℂ)))) {s : ℂ | 1/2 < s.re} :=
    lemma32_regular_euler_product_analyticOnNhd χ
