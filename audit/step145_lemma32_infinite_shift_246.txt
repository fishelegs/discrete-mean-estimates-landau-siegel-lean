lemma lemma32_actual_horizontal_integral_bound {D : ℕ} (χ : RealPrimitiveCharacter D)
    (hD : 1 < D) (t : ℝ) (ht : 1 ≤ |t|) :
    ‖∫ x : ℝ in (-1/4)..1, lemma32CircleIntegrand χ ((x : ℂ)+(t : ℂ)*I)‖ ≤
      (5/4)*lemma32GlobalHeightConstant D/|t|^4 := by
  have hbound : ∀ x ∈ Set.uIoc (-1/4 : ℝ) 1,
      ‖lemma32CircleIntegrand χ ((x : ℂ)+(t : ℂ)*I)‖ ≤ lemma32GlobalHeightConstant D/|t|^4 := by
    intro x hx
    rw [Set.uIoc_of_le (by norm_num : (-1/4 : ℝ) ≤ 1)] at hx
    simpa using lemma32_actual_integrand_height_decay χ hD ((x : ℂ)+(t : ℂ)*I)
      (by simpa using hx.1.le) (by simpa using hx.2) (by simpa using ht)
  have hi := intervalIntegral.norm_integral_le_of_norm_le_const hbound
  calc
    _ ≤ (lemma32GlobalHeightConstant D/|t|^4)*|1-(-1/4 : ℝ)| := hi
    _ = _ := by norm_num;ring

lemma lemma32_tendsto_zero_of_quartic_norm_bound (f : ℝ → ℂ) (C : ℝ)
    (hb : ∀ T : ℝ, 1 ≤ T → ‖f T‖ ≤ C/T^4) : Tendsto f atTop (𝓝 0) := by
  have hi : Tendsto (fun T : ℝ => C/T^4) atTop (𝓝 0) := by
    simpa only [div_eq_mul_inv,mul_zero] using
      (tendsto_inv_atTop_zero.comp (tendsto_pow_atTop (by decide : (4 : ℕ) ≠ 0))).const_mul C
  apply tendsto_zero_iff_norm_tendsto_zero.mpr
  apply squeeze_zero' (Eventually.of_forall (fun T => norm_nonneg _)) _ hi
  filter_upwards [eventually_ge_atTop (1 : ℝ)] with T hT
  exact hb T hT

lemma lemma32_actual_upper_horizontal_decay {D : ℕ} (χ : RealPrimitiveCharacter D) (hD : 1 < D) :
    Tendsto (fun T : ℝ => ∫ x : ℝ in (-1/4)..1,
      lemma32CircleIntegrand χ ((x : ℂ)+(T : ℂ)*I)) atTop (𝓝 0) := by
  apply lemma32_tendsto_zero_of_quartic_norm_bound _ ((5/4)*lemma32GlobalHeightConstant D)
  intro T hT
  have hT0 : 0 ≤ T := by linarith
  simpa only [abs_of_nonneg hT0] using lemma32_actual_horizontal_integral_bound χ hD T
    (by simpa only [abs_of_nonneg hT0] using hT)

lemma lemma32_actual_lower_horizontal_decay {D : ℕ} (χ : RealPrimitiveCharacter D) (hD : 1 < D) :
    Tendsto (fun T : ℝ => ∫ x : ℝ in (-1/4)..1,
      lemma32CircleIntegrand χ ((x : ℂ)-(T : ℂ)*I)) atTop (𝓝 0) := by
  apply lemma32_tendsto_zero_of_quartic_norm_bound _ ((5/4)*lemma32GlobalHeightConstant D)
  intro T hT
  have hT0 : 0 ≤ T := by linarith
  simpa only [Complex.ofReal_neg,neg_mul,← sub_eq_add_neg,abs_neg,abs_of_nonneg hT0] using
    lemma32_actual_horizontal_integral_bound χ hD (-T)
      (by simpa only [abs_neg,abs_of_nonneg hT0] using hT)
