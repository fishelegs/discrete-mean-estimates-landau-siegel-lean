      2 * (∑ n ∈ Finset.Ioc Y N, (lemma23NuArithmeticFunction χ n).re * (n : ℝ)⁻¹) *
        (∑ n ∈ Finset.Icc 1 N, (lemma23NuArithmeticFunction χ n).re * (n : ℝ)⁻¹) :=
  lemma31_actual_square_weighted_tail_le χ Y N

      (dirichletLFunction χ (1 : ℂ)).re| ≤ 2 * (D : ℝ) / N :=
