lemma lemma32_actual_CRT_conductor_dvd_lcm {m n : ℕ} [NeZero m] [NeZero n]
    (h : m.Coprime n) (χ : DirichletCharacter ℂ (m*n)) :
    χ.conductor ∣ (lemma32CRTCharacterLeft h χ).conductor.lcm
      (lemma32CRTCharacterRight h χ).conductor := by
  letI : NeZero (m*n) := ⟨Nat.mul_ne_zero (NeZero.ne m) (NeZero.ne n)⟩
  have hl := lemma32_changeLevel_conductor_dvd (Nat.dvd_mul_right m n)
    (lemma32CRTCharacterLeft h χ)
  have hr := lemma32_changeLevel_conductor_dvd (Nat.dvd_mul_left n m)
    (lemma32CRTCharacterRight h χ)
  have hc := DirichletCharacter.conductor_mul_dvd_lcm_conductor
    (DirichletCharacter.changeLevel (Nat.dvd_mul_right m n) (lemma32CRTCharacterLeft h χ))
    (DirichletCharacter.changeLevel (Nat.dvd_mul_left n m) (lemma32CRTCharacterRight h χ))
  rw [← lemma32_actual_CRT_changeLevel_product h χ] at hc
  exact hc.trans (Nat.lcm_dvd (hl.trans (Nat.dvd_lcm_left _ _))
    (hr.trans (Nat.dvd_lcm_right _ _)))

lemma lemma32_actual_primitive_CRT_characters {m n : ℕ} [NeZero m] [NeZero n]
    (h : m.Coprime n) (χ : DirichletCharacter ℂ (m*n)) (hp : χ.IsPrimitive) :
    (lemma32CRTCharacterLeft h χ).IsPrimitive ∧ (lemma32CRTCharacterRight h χ).IsPrimitive := by
  letI : NeZero (m*n) := ⟨Nat.mul_ne_zero (NeZero.ne m) (NeZero.ne n)⟩
  let d := (lemma32CRTCharacterLeft h χ).conductor
  let e := (lemma32CRTCharacterRight h χ).conductor
  have hd : d ∣ m := (lemma32CRTCharacterLeft h χ).conductor_dvd_level
  have he : e ∣ n := (lemma32CRTCharacterRight h χ).conductor_dvd_level
  have hdp : 0 < d := Nat.pos_of_ne_zero (lemma32CRTCharacterLeft h χ).conductor_ne_zero
  have hep : 0 < e := Nat.pos_of_ne_zero (lemma32CRTCharacterRight h χ).conductor_ne_zero
  have hmp : 0 < m := Nat.pos_of_ne_zero (NeZero.ne m)
  have hnp : 0 < n := Nat.pos_of_ne_zero (NeZero.ne n)
  have hc : m*n ∣ d.lcm e := by
    have hc := lemma32_actual_CRT_conductor_dvd_lcm h χ
    rw [hp] at hc
    exact hc
  constructor
  · change d=m
    apply Nat.le_antisymm (Nat.le_of_dvd hmp hd)
    have hdiv : m*n ∣ d*n := hc.trans
      (Nat.lcm_dvd (Nat.dvd_mul_right d n) (he.trans (Nat.dvd_mul_left n d)))
    have hle := Nat.le_of_dvd (Nat.mul_pos hdp hnp) hdiv
    by_contra hnot
    exact (not_lt_of_ge hle) (Nat.mul_lt_mul_of_pos_right (Nat.lt_of_not_ge hnot) hnp)
  · change e=n
    apply Nat.le_antisymm (Nat.le_of_dvd hnp he)
    have hdiv : m*n ∣ m*e := hc.trans
      (Nat.lcm_dvd (hd.trans (Nat.dvd_mul_right m e)) (Nat.dvd_mul_left e m))
    have hle := Nat.le_of_dvd (Nat.mul_pos hmp hep) hdiv
    by_contra hnot
    exact (not_lt_of_ge hle) (Nat.mul_lt_mul_of_pos_left (Nat.lt_of_not_ge hnot) hmp)

end ZhangLS.Spec
